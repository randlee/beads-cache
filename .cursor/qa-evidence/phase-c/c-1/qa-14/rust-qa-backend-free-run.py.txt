import json, os, pathlib, shutil, subprocess, sys, time

root=pathlib.Path('/tmp/phase-c-execution/c-1/qa-14')
original_path=os.environ['PATH']
original_dirs=original_path.split(os.pathsep)
removed=[]
retained=[]
for directory in original_dirs:
    hits=[tool for tool in ('bd','dolt') if shutil.which(tool,path=directory)]
    if hits: removed.append(dict(directory=directory,exposes=hits))
    else: retained.append(directory)
env=dict(os.environ,PATH=os.pathsep.join(retained),CARGO_INCREMENTAL='0',CARGO_TARGET_DIR='/tmp/phase-c-execution/c-1/restart-target',RUNNER_TEMP=str(root))
assert shutil.which('bd',path=env['PATH']) is None
assert shutil.which('dolt',path=env['PATH']) is None
tools={name:shutil.which(name,path=env['PATH']) for name in ('cargo','rustc','cc','git','python3')}
assert all(tools.values()),tools
assert tools['cargo']==shutil.which('cargo',path=original_path)
python=str(pathlib.Path(sys.executable).resolve())
assert pathlib.Path(python).is_file()
environment=dict(original_PATH=original_path,backend_free_PATH=env['PATH'],removed_directories=removed,retained_executables=tools,absolute_python=python,bd_resolves=False,dolt_resolves=False,scope='Process-local PATH only; remove only directories exposing bd/dolt; no source/timer/global edits')
(root/'rust-qa-backend-free-environment.json').write_text(json.dumps(environment,indent=2)+'\n')
selected=next(row['command'] for row in json.loads((root/'rust-qa-receipts.json').read_text()) if row['name']=='sync-tests')
commands=[('no-backend-path-gated',[python,'-I','scripts/check_no_backend.py','path']),('sync-tests-backend-free',selected)]
receipts=[]
for name,command in commands:
    print('START',name,flush=True);start=time.monotonic()
    log_path=root/(name+'.log')
    with log_path.open('w') as log: result=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT)
    receipts.append(dict(name=name,command=command,exit=result.returncode,expected_exit=0,seconds=round(time.monotonic()-start,3),log=str(log_path),execution_context='One parent-authorized native process-local backend-free PATH qualification',cwd=os.getcwd(),environment_receipt=str(root/'rust-qa-backend-free-environment.json')))
    (root/'rust-qa-backend-free-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
    print('END',name,result.returncode,flush=True)
    if result.returncode: break
