import json, os, pathlib, subprocess, time

root=pathlib.Path('/tmp/phase-c-execution/c-1/qa-14')
env=dict(os.environ,CARGO_INCREMENTAL='0',CARGO_TARGET_DIR='/tmp/phase-c-execution/c-1/restart-target',RUNNER_TEMP=str(root))
commands=[
    ('isolated-graph-native',['python3','-I','crates/bd-cache-sync/tests/check_consumer_graph.py'],0),
    ('negative-server-native',['python3','-I','crates/bd-cache-sync/tests/check_consumer_graph.py','--negative-server'],1),
]
receipts=[]
for name,command,expected in commands:
    print('START',name,flush=True); start=time.monotonic()
    log_path=root/(name+'.log')
    with log_path.open('w') as log: result=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT)
    receipts.append(dict(name=name,command=command,exit=result.returncode,expected_exit=expected,seconds=round(time.monotonic()-start,3),log=str(log_path),execution_context='one authorized identical native run after sandbox Cargo metadata failure; no source/timing/retry changes',cwd=os.getcwd()))
    (root/'rust-qa-native-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
    print('END',name,result.returncode,flush=True)
