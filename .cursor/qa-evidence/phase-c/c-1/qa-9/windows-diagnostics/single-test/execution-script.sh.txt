#!/bin/bash
# DRAFT ONLY: parent must authorize execution after inspecting terminal C2 output.
set -euo pipefail
if [ "$#" -ne 1 ]; then
  echo "usage: $0 /absolute/path/to/authorized/isolated/worktree" >&2
  exit 2
fi
worktree_dir="$1"
expected_source=e86d9da92a8c8c9b787c780e29bc54e417680d71
expected_branch=codex/c2-windows-store-phase-observation
draft_dir="$(CDPATH= cd -- "$(dirname -- "$0")" && pwd)"
test "$(git -C "$worktree_dir" branch --show-current)" = "$expected_branch"
test "$(git -C "$worktree_dir" rev-parse HEAD)" = "$expected_source"
test -z "$(git -C "$worktree_dir" status --porcelain)"
cp "$draft_dir/one-time-windows-store-phase-observation.yml" "$worktree_dir/.github/workflows/one-time-windows-store-phase-observation.yml"
git -C "$worktree_dir" add .github/workflows/one-time-windows-store-phase-observation.yml
git -C "$worktree_dir" diff --cached --check
test "$(git -C "$worktree_dir" diff --cached --name-only)" = .github/workflows/one-time-windows-store-phase-observation.yml
git -C "$worktree_dir" commit -m 'Observe one Windows store fixture with captured phases'
git -C "$worktree_dir" push -u origin "$expected_branch"
