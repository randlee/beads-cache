# One-time diagnostic branch only. Never merge into integration.
name: one-time Windows store phase observation
on:
  push:
    branches: [codex/c2-windows-store-phase-observation]
permissions:
  contents: read
jobs:
  observe:
    runs-on: windows-latest
    timeout-minutes: 15
    defaults:
      run:
        shell: bash
    env:
      CARGO_TERM_COLOR: always
      CARGO_INCREMENTAL: '0'
      RUSTFLAGS: -D warnings
      EXPECTED_SOURCE: e86d9da92a8c8c9b787c780e29bc54e417680d71
    steps:
      - uses: actions/checkout@v4
        with:
          ref: e86d9da92a8c8c9b787c780e29bc54e417680d71
      - name: Verify immutable test source
        run: |
          test "$(git rev-parse HEAD)" = "$EXPECTED_SOURCE"
          git log -1 --format=%H
          git rev-parse HEAD:Cargo.lock HEAD:crates/bd-cache-store/src/schedule.rs HEAD:crates/bd-cache-store/tests/support/fixture_child.rs
      - name: Install the pinned toolchain
        run: rustup show active-toolchain || rustup toolchain install
      - uses: Swatinem/rust-cache@v2
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - name: No bd or dolt on PATH
        run: python -I scripts/check_no_backend.py path
      - name: Record selected store SQLite feature context
        run: cargo tree --locked -p bd-cache-store --all-features --target x86_64-pc-windows-msvc --edges features --invert rusqlite
      - name: Observe exactly one supervisor negative test
        run: |
          set -o pipefail
          cargo test --locked -p bd-cache-store --lib --all-features schedule::tests::shutdown_fixture_supervisor_kills_and_reaps_each_stalled_cleanup -- --exact --nocapture 2>&1 | tee "$RUNNER_TEMP/windows-store-phase-observation.log"
      - name: Preserve raw focused output on success or failure
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: windows-store-phase-observation-e86d9da
          path: ${{ runner.temp }}/windows-store-phase-observation.log
          if-no-files-found: warn
          retention-days: 7
