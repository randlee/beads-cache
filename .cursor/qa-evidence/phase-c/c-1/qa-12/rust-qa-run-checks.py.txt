import subprocess, os, json, time
from pathlib import Path
root=Path("/tmp/phase-c-execution/c-1/qa-12")
wd="/Users/randlee/github/beads-cache-worktrees/sprint/c-1-runtime-migration"
env=os.environ.copy(); env.update(CARGO_INCREMENTAL="0",CARGO_TARGET_DIR="/tmp/phase-c-execution/c-1/restart-target")
checks=[("fmt",["cargo","fmt","--all","--check"]),("clippy",["cargo","clippy","-p","bd-cache-store","-p","bd-cache-sync","--all-targets","--all-features","--","-D","warnings"]),("store",["cargo","test","-p","bd-cache-store","--lib","--all-features"]),("cron_hurry",["cargo","test","-p","bd-cache-sync","--test","cron_hurry","--all-features"])]
receipts=[]
for name,cmd in checks:
    start=time.time(); print(json.dumps({"start":name,"command":cmd}),flush=True)
    log=root/("rust-qa-"+name+".log")
    with log.open("w") as out:
        result=subprocess.run(cmd,cwd=wd,env=env,stdout=out,stderr=subprocess.STDOUT)
    rec={"name":name,"command":cmd,"exit_code":result.returncode,"elapsed_seconds":time.time()-start,"log":str(log),"native_context":True}; receipts.append(rec)
    (root/"rust-qa-check-receipts.json").write_text(json.dumps(receipts,indent=2)+"\n")
    print(json.dumps(rec),flush=True)
(root/"rust-qa-cargo-release.json").write_text(json.dumps({"released":True,"checks_finished":True,"active_child":False,"time":time.time(),"target_dir":env["CARGO_TARGET_DIR"]},indent=2)+"\n")
print("CARGO_SLOT_RELEASED",flush=True)
