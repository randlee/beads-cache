python3 - "$RUNNER_TEMP/metadata.json" docs/architecture.md <<'PY'
"""Fail on any workspace-crate edge the Boundaries table does not allow.

Every dependency kind counts (normal, dev, build). A crate missing from
the table fails too, so a new crate must be added to the table first.
A crate's dev-dependency on itself (to enable its own test features)
is not an edge.

The table covers workspace edges only, so a second check rejects a
direct dependency (any kind) of bd-cache-api or bd-cache-py on
sc-command or clap (contract section 9, sprint b-2).
"""
import json, re, sys

meta = json.load(open(sys.argv[1]))
doc = open(sys.argv[2], encoding="utf-8").read()
section = doc.split("## Boundaries and Dependencies", 1)[1].split("\n## ", 1)[0]
allowed = {}
for line in section.splitlines():
    cells = [c.strip() for c in line.strip().strip("|").split("|")]
    if len(cells) != 2 or not cells[0].startswith("`"):
        continue
    crate = cells[0].strip("`")
    rule = cells[1].split("(")[0]  # parenthesised notes are commentary
    allowed[crate] = set(re.findall(r"`([^`]+)`", rule))
members = {p["name"] for p in meta["packages"]}
errors = []
for pkg in meta["packages"]:
    name = pkg["name"]
    if name not in allowed:
        errors.append(f"{name}: not in the Boundaries table of docs/architecture.md")
        continue
    for dep in pkg["dependencies"]:
        target = dep["name"]
        if target in members and target != name and target not in allowed[name]:
            kind = dep.get("kind") or "normal"
            errors.append(f"{name} -> {target} ({kind}): not allowed")
no_cli = {"bd-cache-api", "bd-cache-py"}
cli_only = {"sc-command", "clap"}
for pkg in meta["packages"]:
    if pkg["name"] not in no_cli:
        continue
    for dep in pkg["dependencies"]:
        names = {dep["name"], dep.get("rename") or dep["name"]}
        if names & cli_only:
            kind = dep.get("kind") or "normal"
            errors.append(f"{pkg['name']} -> {dep['name']} ({kind}): no sc-command or clap (contract section 9)")
for e in errors:
    print(f"::error::{e}")
if errors:
    sys.exit(1)
print(f"boundaries ok: {sorted(members)}")
PY
