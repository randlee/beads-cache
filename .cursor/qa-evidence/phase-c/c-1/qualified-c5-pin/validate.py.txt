import json,os,subprocess,time
from pathlib import Path
root=Path('/tmp/phase-c-execution/c-1/qualified-c5-pin')
env=dict(os.environ,CARGO_INCREMENTAL='0',CARGO_TARGET_DIR='/tmp/phase-c-execution/c-1/restart-target',RUNNER_TEMP=str(root))
commands=[
('store-tests',['cargo','test','--locked','-p','bd-cache-store','--lib','--all-features','--','--nocapture']),
('sync-tests',['cargo','test','--locked','-p','bd-cache-sync','--all-features','--','--nocapture']),
('cli-contracts',['cargo','test','--locked','-p','bd-cache-cli','--test','fixtures','--test','cli','--all-features','--','--nocapture']),
('direct-boundary',['bash',str(root/'direct-boundary-workflow.sh')]),
('postpush-graph-unit',['python3','-I','-m','unittest','discover','-s','crates/bd-cache-sync/tests','-p','test_consumer_graph.py','-v']),
('postpush-isolated-graph',['python3','-I','crates/bd-cache-sync/tests/check_consumer_graph.py']),
('negative-server',['python3','-I','crates/bd-cache-sync/tests/check_consumer_graph.py','--negative-server']),
('no-backend',['python3','-I','scripts/check_no_backend.py','graph']),
('fmt-check',['cargo','fmt','--all','--check']),
('diff-check',['git','diff','--check']),
]
receipts=[]
for name,command in commands:
 print('START',name,flush=True);start=time.monotonic()
 with (root/f'{name}.log').open('w') as log:
  run=subprocess.run(command,env=env,stdout=log,stderr=subprocess.STDOUT)
 expected=1 if name=='negative-server' else 0
 receipts.append(dict(name=name,command=command,exit=run.returncode,expected_exit=expected,elapsed_seconds=round(time.monotonic()-start,3),log=str(root/f'{name}.log')))
 (root/'validation-receipts.json').write_text(json.dumps(receipts,indent=2)+'\n')
 print('END',name,'exit',run.returncode,'expected',expected,flush=True)
 if run.returncode!=expected:raise SystemExit(run.returncode or 1)
