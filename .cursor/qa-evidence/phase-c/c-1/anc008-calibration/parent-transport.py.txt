import concurrent.futures,hashlib,json,os,pathlib,ssl,time,urllib.request,urllib.error,certifi
root=pathlib.Path('/tmp/phase-c-execution/c-1/qa-15/anc008-calibration')
def canon(v):return json.dumps(v,sort_keys=True,separators=(',',':'),ensure_ascii=False).encode()
def sha(b):return hashlib.sha256(b).hexdigest()
m=json.loads((root/'baseline/manifest.json').read_text())
assert m['endpoint']=='https://api.typesafe.ai/v1/systemone' and m['method']=='POST'
assert len(m['entries'])==28 and m['limits']['max_parallel']==4
assert sha(canon(json.loads(pathlib.Path(m['oracle_path']).read_text())))==m['oracle_sha256']
assert json.loads((root/'parent-envelope-validation.json').read_text())['data']['valid']
assert json.loads((root/'preparation-validation.json').read_text())['network_calls']==0
key=os.environ.get('TYPESAFE_API_KEY','').strip();assert key and key.isascii() and not any(c.isspace() for c in key)
ctx=ssl.create_default_context(cafile=certifi.where());entries=[]
for e in m['entries']:
 assert e['candidate_id']=='c00'
 data=canon(json.loads(pathlib.Path(e['request_path']).read_text()));assert sha(data)==e['canonical_request_sha256']
 target=pathlib.Path(e['response_path']);assert not target.exists(),str(target)
 entries.append((e,data))
receipts=root/'baseline/transport';receipts.mkdir(exist_ok=True)
def send(pair):
 e,data=pair;target=pathlib.Path(e['response_path']);target.parent.mkdir(parents=True,exist_ok=True)
 records=[]
 for attempt in [1,2]:
  rec={'correlation_id':e['correlation_id'],'attempt':attempt,'canonical_request_sha256':sha(data),'candidate_id':'c00'}
  start=time.time();raw=None;retry=False
  try:
   req=urllib.request.Request(m['endpoint'],data=data,headers={'Authorization':'Bearer '+key,'Content-Type':'application/json'},method='POST')
   with urllib.request.urlopen(req,timeout=30,context=ctx) as response:
    raw=response.read();rec['http_status']=response.status
   raw=raw.replace(key.encode(),b'<redacted>');json.loads(raw);rec['success']=True
  except urllib.error.HTTPError as err:
   rec.update(success=False,http_status=err.code,error_code='HTTP_'+str(err.code));retry=err.code==429 or 500<=err.code<600
  except (urllib.error.URLError,TimeoutError) as err:
   rec.update(success=False,error_code=type(err).__name__);retry=True
  except Exception as err:
   rec.update(success=False,error_code=type(err).__name__)
  rec['elapsed_seconds']=time.time()-start
  if raw is not None:
   ap=receipts/(e['correlation_id']+'.attempt'+str(attempt)+'.raw.json');ap.write_bytes(raw);rec['raw_path']=str(ap);rec['raw_sha256']=sha(raw)
  (receipts/(e['correlation_id']+'.attempt'+str(attempt)+'.receipt.json')).write_text(json.dumps(rec,indent=2)+'\n');records.append(rec)
  if rec['success']:
   target.write_bytes(raw);break
  if not retry:break
 return {'correlation_id':e['correlation_id'],'success':records[-1]['success'],'attempts':records}
start=time.time()
with concurrent.futures.ThreadPoolExecutor(max_workers=4) as executor:out=list(executor.map(send,entries))
summary={'oracle_sha256':m['oracle_sha256'],'candidate':'c00','requests':len(out),'successful':sum(x['success'] for x in out),'failed':sum(not x['success'] for x in out),'elapsed_seconds':time.time()-start,'certificate_verification_enabled':True,'credential_source':'existing environment; not persisted','results':out}
(root/'baseline/parent-transport-receipt.json').write_text(json.dumps(summary,indent=2)+'\n');print(json.dumps({k:v for k,v in summary.items() if k!='results'}))
